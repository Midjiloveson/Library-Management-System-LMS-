    </Routes>
  );
}

// Automatically attach JWT token to every request
API.interceptors.request.use((req) => {
  const token = localStorage.getItem('token');
  if (token) {
    req.headers.Authorization = `Bearer ${token}`;
  }
  return req;
});

import { createContext, useState, useEffect } from 'react';

export const AuthContext = createContext();

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const savedUser = localStorage.getItem('userInfo');
    if (savedUser) setUser(JSON.parse(savedUser));
    setLoading(false);
  }, []);

  const login = (data) => {
    localStorage.setItem('token', data.token);
    localStorage.setItem('userInfo', JSON.stringify(data.user));
    setUser(data.user);
  };

  const logout = () => {
    localStorage.clear();
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ user, login, logout, loading }}>
      {children}
    </AuthContext.Provider>
  );
};

import { useState, useEffect, useContext, useMemo } from 'react';
import { 
  AppShell, Group, Title, Text, Button, 
  Image, Badge, Container, TextInput, Loader, Center, Stack, Anchor 
} from '@mantine/core';
import { 
  IconSearch, IconLogout, IconLibrary, 
  IconChevronRight, IconBookOff 
} from '@tabler/icons-react';
import { AuthContext } from '../context/AuthContext';
import API from '../api/axiosInstance';

export default function Dashboard() {
  const [books, setBooks] = useState([]);
  const [loading, setLoading] = useState(true);
  const [search, setSearch] = useState('');
  const { user, logout } = useContext(AuthContext);

  const fetchBooks = async () => {
    try {
      setLoading(true);
      const { data } = await API.get('/books');
      setBooks(data);
    } catch (err) {
      console.error("Library Error:", err);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => { fetchBooks(); }, []);

  // 1. UNIVERSAL SEARCH LOGIC (Title, Author, Category, ISBN)
  const searchResults = useMemo(() => {
    if (!search) return [];
    const query = search.toLowerCase().trim();

    return books.filter(book => {
      const titleMatch    = book.title?.toLowerCase().includes(query);
      const authorMatch   = book.author?.toLowerCase().includes(query);
      const categoryMatch = book.category?.toLowerCase().includes(query);
      const isbnMatch     = book.isbn?.includes(query); 

      return titleMatch || authorMatch || categoryMatch || isbnMatch;
    });
  }, [search, books]);

  // 2. GENRE GROUPING LOGIC
  const groupedBooks = useMemo(() => {
    const groups = {};
    books.forEach(book => {
      const genre = book.category || 'General Collection';
      if (!groups[genre]) groups[genre] = [];
      groups[genre].push(book);
    });
    return groups;
  }, [books]);

  const handleSeeMore = (genreName) => {
    setSearch(genreName);
    window.scrollTo({ top: 0, behavior: 'smooth' });
  };

  return (
    <AppShell 
      header={{ height: 70 }} 
      padding="md" 
      className="bg-[#fdfaf6]"
    >
      {/* RESPONSIVE HEADER WITH ROLE LOGIC */}
      <AppShell.Header className="border-b border-slate-200 bg-white/90 backdrop-blur-sm px-2 sm:px-4">
        <Container size="lg" className="h-full px-0 sm:px-3">
          <Group justify="space-between" align="center" className="h-full flex-nowrap">
            
            {/* LEFT: Logo & Title */}
            <Group gap="xs" className="flex-nowrap shrink">
              <div className="bg-slate-900 p-1.5 rounded-lg shrink-0">
                <IconLibrary size={18} color="white" stroke={1.5} />
              </div>
              <Title 
                order={3} 
                className="font-serif italic text-slate-800 truncate text-lg sm:text-xl"
              >
                The Archives
              </Title>
            </Group>
            
            {/* RIGHT: User & Exit */}
            <Group gap="xs" className="flex-nowrap">
              <div className="hidden md:flex flex-col items-end shrink-0">
                <Text size="xs" c="dimmed" fw={700} className="uppercase tracking-widest">
                  {user?.role === 'admin' ? 'Librarian' : 'Member'}
                </Text>
                <Text size="sm" fw={600}>{user?.firstName}</Text>
              </div>

              <Button 
                variant="light" 
                color="red" 
                onClick={logout} 
                size="xs"
                className="px-2 sm:px-3"
                leftSection={<IconLogout size={14} />}
              >
                <span className="hidden sm:inline">Exit Archives</span>
                <span className="inline sm:hidden">Exit</span>
              </Button>
            </Group>

          </Group>
        </Container>
      </AppShell.Header>

      <AppShell.Main>
        {/* STICKY FLOATING SEARCH BAR */}
        <div className="sticky top-[70px] z-20 bg-[#fdfaf6]/95 backdrop-blur-md border-b border-slate-200 py-4 sm:py-6 mb-6 sm:mb-10 shadow-sm">
          <Container size="sm">
            <TextInput
              size="lg"
              radius="xl"
              placeholder="Search Title, Author, Genre, or ISBN..."
              leftSection={<IconSearch size={20} stroke={1.5} />}
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              className="shadow-md"
              rightSection={search && (
                <Anchor component="button" size="xs" onClick={() => setSearch('')} pr="md">
                  Clear
                </Anchor>
              )}
            />
          </Container>
        </div>

        <Container size="lg">
          {loading ? (
            <Center h={300}><Loader color="dark" type="dots" /></Center>
          ) : search ? (
            /* SEARCH RESULTS VIEW */
            <>
              <Group justify="space-between" mb="xl" px="xs">
                <Title order={2} className="font-serif italic text-slate-800 text-xl sm:text-2xl">
                  Results for "{search}"
                </Title>
                <Text size="xs" c="dimmed" fw={700}>{searchResults.length} FOUND</Text>
              </Group>
              
              {searchResults.length > 0 ? (
                <div className="grid grid-cols-2 md:grid-cols-4 lg:grid-cols-6 gap-4 sm:gap-8 pb-20">
                  {searchResults.map(book => <BookItem key={book._id} book={book} />)}
                </div>
              ) : (
                <Center h={200}>
                  <Stack align="center" gap="xs">
                    <IconBookOff size={48} color="#cbd5e1" />
                    <Text c="dimmed" fs="italic">No records match your query.</Text>
                  </Stack>
                </Center>
              )}
            </>
          ) : (
            /* GROUPED GENRE VIEW */
            Object.keys(groupedBooks).map(genre => (
              <section key={genre} className="mb-12 sm:mb-20">
                <Group justify="space-between" mb="lg" px="xs" className="border-b border-slate-200 pb-2">
                  <Title order={3} className="font-serif text-slate-700 text-lg sm:text-2xl">{genre}</Title>
                  <Button 
                    variant="subtle" 
                    color="indigo" 
                    size="xs"
                    rightSection={<IconChevronRight size={14} />}
                    onClick={() => handleSeeMore(genre)}
                  >
                    See More
                  </Button>
                </Group>
                
                <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-6 gap-4 sm:gap-8">
                  {groupedBooks[genre].slice(0, 6).map(book => (
                    <BookItem key={book._id} book={book} />
                  ))}
                </div>
              </section>
            ))
          )}
        </Container>
      </AppShell.Main>
    </AppShell>
  );
}

/* REUSABLE BOOK COMPONENT */
function BookItem({ book }) {
  return (
    <div className="group cursor-pointer">
      <div className="relative aspect-[2/3] mb-3 overflow-hidden rounded-sm shadow-sm group-hover:shadow-xl transition-all duration-300 transform group-hover:-translate-y-1 border-l-[4px] sm:border-l-[6px] border-slate-900/10">
        <Image
          src={book.coverImage || 'https://images.unsplash.com/photo-1543002588-bfa74002ed7e?q=80&w=400'}

          {book.author}
        </Text>
      </div>
    </div>
  );
}

const seedFromFile = async () => {

    console.log("Connecting to MongoDB...");
    await mongoose.connect(process.env.MONGO_URI);
    
    // 1. Wipe the current collection to ensure a clean start
    await Book.deleteMany({});
    console.log("Database cleared.");

    // 2. Resolve the path to books.json
    const filePath = path.join(__dirname, 'books.json');

    // 3. Check if the file exists before reading
    if (!fs.existsSync(filePath)) {
      throw new Error("books.json not found! Please create it in the server folder.");
    }

    // 4. Read and parse the JSON data
    const rawData = JSON.parse(fs.readFileSync(filePath, 'utf-8'));
    
    // 5. Map defaults to all books
    const preparedData = rawData.map(book => ({
      ...book,
      isAvailable: book.isAvailable ?? true, // Keep original if exists, otherwise true
      isDeleted: false,      // All seeded books start as active
      holds: [],             // Empty waitlist
      borrowedBy: null,      // No one has it yet
      dueDate: null,         // No due date yet
      holdStartedAt: null    // No active 48hr timers
    }));

    // 6. Bulk insert into Atlas
    await Book.insertMany(preparedData);
    
    console.log(`\n✅ SUCCESS!`);
    console.log(`Synchronized ${preparedData.length} books from books.json to MongoDB with full system defaults.`);
    
    process.exit();
  } catch (err) {
    console.error("❌ Seeding Error:", err.message);
    process.exit(1);
  }
};

const User = require('../models/User');
const Book = require('../models/Book');
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');

// Register a new user
exports.register = async (req, res) => {
  const { firstName, lastName, email, phone, password } = req.body;
  try {
    let user = await User.findOne({ email });
    if (user) return res.status(400).json({ message: "User already exists" });

    user = new User({ firstName, lastName, email, phone, password });

    const salt = await bcrypt.genSalt(10);
    user.password = await bcrypt.hash(password, salt);

    await user.save();
    res.status(201).json({ message: "User registered successfully" });
  } catch (err) {
    res.status(500).json({ message: "Server error", error: err.message });
  }
};

// Login user
exports.login = async (req, res) => {
  const { email, password } = req.body;
  try {
    const user = await User.findOne({ email });
    if (!user) return res.status(400).json({ message: "Invalid Credentials" });

    const isMatch = await bcrypt.compare(password, user.password);
    if (!isMatch) return res.status(400).json({ message: "Invalid Credentials" });

    const token = jwt.sign(
      { id: user._id, role: user.role }, 
      process.env.JWT_SECRET || 'your_fallback_secret', 
      { expiresIn: '1d' }
    );

    res.json({
      token,
      user: { id: user._id, firstName: user.firstName, role: user.role }
    });
  } catch (err) {
    res.status(500).json({ message: "Server error" });
  }
};

// Delete account logic
exports.deleteAccount = async (req, res) => {
  try {
    const user = await User.findById(req.user.id);
    if (!user) return res.status(404).json({ message: "User not found" });

    // 1. Block if they have books
    if (user.borrowedBooks && user.borrowedBooks.length > 0) {
      return res.status(400).json({ 
        message: "Account cannot be deleted. You must return all borrowed books first!" 
      });
    }

    // 2. Cleanup Hold Queue
    await Book.updateMany(
      { holds: req.user.id },
      { $pull: { holds: req.user.id } }
    );

    // 3. Delete User
    await User.findByIdAndDelete(req.user.id);
    res.json({ message: "Account deleted successfully." });
  } catch (err) {
    res.status(500).json({ message: "Server error", error: err.message });
  }
};

exports.borrowBook = async (req, res) => {
  try {
    const book = await Book.findById(req.params.id);
    const isAdmin = req.user.role === 'admin';
    const effectiveUserId = (isAdmin && req.body.targetUserId) ? req.body.targetUserId : req.user.id;
    
    const user = await User.findById(effectiveUserId);
    if (!book || book.isDeleted) return res.status(404).json({ message: "Book not found" });
    if (!user) return res.status(404).json({ message: "User not found" });

    if (book.holds && book.holds.length > 0) {
      const nextInLine = book.holds[0].toString();
      if (nextInLine !== effectiveUserId) {
        return res.status(403).json({ message: "This book is reserved for someone else." });
      }
      book.holds.shift();
      book.holdStartedAt = null; 
    } else if (!book.isAvailable) {
      return res.status(400).json({ message: "Book is already checked out" });
    }

    if (user.borrowedBooks.length >= 10) return res.status(400).json({ message: "User checkout limit (10) reached." });

    const dueDate = new Date();
    dueDate.setDate(dueDate.getDate() + 14);

    book.isAvailable = false;
    book.dueDate = dueDate;
    book.borrowedBy = user._id;
    await book.save();

    user.borrowedBooks.push(book._id);
    await user.save();

    res.json({ message: `Checked out "${book.title}" for ${user.firstName}. Due: ${dueDate.toDateString()}` });
  } catch (err) {
    res.status(500).json({ message: "Checkout failed", error: err.message });
  }
};

exports.placeHold = async (req, res) => {
  try {
    const book = await Book.findById(req.params.id);
    const isAdmin = req.user.role === 'admin';
    const effectiveUserId = (isAdmin && req.body.targetUserId) ? req.body.targetUserId : req.user.id;

    if (!book || book.isDeleted) return res.status(404).json({ message: "Book not found" });

    // FIX: Prevent user from holding a book they already have checked out
    if (book.borrowedBy && book.borrowedBy.toString() === effectiveUserId) {
      return res.status(400).json({ 
        message: "You currently have this book checked out. If you need it longer, please try to 'Renew' it instead." 
      });
    }

    if (book.isAvailable && (!book.holds || book.holds.length === 0)) {
      return res.status(400).json({ message: "Book is available for direct checkout." });
    }
    
    const alreadyWaiting = book.holds.some(holdId => holdId.toString() === effectiveUserId);
    if (alreadyWaiting) return res.status(400).json({ message: "User is already on the waitlist." });

    book.holds.push(effectiveUserId);
    await book.save();
    res.json({ message: `Hold placed. Position: ${book.holds.length}` });
  } catch (err) {
    res.status(500).json({ message: "Hold failed", error: err.message });
  }
};

exports.renewBook = async (req, res) => {
  try {
    const book = await Book.findById(req.params.id);

    const newDueDate = new Date(); 
    newDueDate.setDate(newDueDate.getDate() + 14);
    book.dueDate = newDueDate;
    await book.save();

    res.json({ message: `Renewed. New due date: ${newDueDate.toDateString()}` });
  } catch (err) {

  }
};

exports.returnBook = async (req, res) => {
  try {
    const book = await Book.findById(req.params.id);
    if (!book || (book.isAvailable && book.holds.length === 0)) {
      return res.status(400).json({ message: "Book is already in the library" });
    }

    const isAdmin = req.user.role === 'admin';
    const isBorrower = book.borrowedBy && book.borrowedBy.toString() === req.user.id;

    if (!isBorrower && !isAdmin) return res.status(403).json({ message: "Unauthorized." });

    const userId = book.borrowedBy;
    book.borrowedBy = null;
    book.dueDate = null;

    if (book.holds.length > 0) {
      book.isAvailable = false;
      book.holdStartedAt = new Date(); 
    } else {
      book.isAvailable = true;
      book.holdStartedAt = null;
    }

    await book.save();

    if (userId) {
      await User.findByIdAndUpdate(userId, { 
        $pull: { borrowedBooks: req.params.id } 
      });
    }

    res.json({ message: "Return processed successfully." });
  } catch (err) {
    res.status(500).json({ message: "Return failed" });
  }
};

const jwt = require('jsonwebtoken');

const protect = (req, res, next) => {
  // 1. Get the full header
  const authHeader = req.header('Authorization');

  // 2. Check if header exists and starts with "Bearer "
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ message: "No token, authorization denied" });
  }

  try {
    // 3. Extract just the token (the part after "Bearer ")
    const token = authHeader.split(' ')[1];

    const decoded = jwt.verify(token, process.env.JWT_SECRET || 'p1DajAoimN3beDcP55uAMLXyMzFfgenrpjs4RQoaXwF5sfqpbBn2BDntybMbgy');
    
    req.user = decoded; 
    next();
  } catch (err) {
    console.error("JWT Error:", err.message); 
    res.status(401).json({ 
      message: "Token is not valid", 
      reason: err.message 
    });
  }
};
const adminOnly = (req, res, next) => {
  if (req.user && req.user.role === 'admin') {
    next(); // User is admin, let them through
  } else {
    res.status(403).json({ message: "Access denied: Admins only" });
  }
};

module.exports = { protect, adminOnly };
